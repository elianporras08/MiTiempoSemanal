package pe.edu.instituto.ponce.mitiemposemanal

import android.os.Bundle
import android.widget.ArrayAdapter
import android.widget.AutoCompleteTextView
import android.widget.Button
import android.widget.TextView
import android.widget.Toast
import androidx.appcompat.app.AlertDialog
import androidx.appcompat.app.AppCompatActivity
import androidx.recyclerview.widget.LinearLayoutManager
import androidx.recyclerview.widget.RecyclerView
import java.time.Duration
import java.time.LocalDateTime
import java.time.format.DateTimeFormatter

class CalendarioActivity : AppCompatActivity() {

    private lateinit var dbManager: DatabaseManager
    private lateinit var etBuscarTarea: AutoCompleteTextView
    private lateinit var btnIniciar: Button
    private lateinit var btnFinalizar: Button
    private lateinit var rvEventos: RecyclerView
    private lateinit var eventoAdapter: EventoAdapter

    private var eventoEnCursoId: Long = -1L
    private val prefsName = "MiTiempoPrefs"
    private val keyEventoId = "evento_en_curso_id"

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_calendario)

        dbManager = DatabaseManager(this)

        etBuscarTarea = findViewById(R.id.etBuscarTarea)
        btnIniciar = findViewById(R.id.btnIniciar)
        btnFinalizar = findViewById(R.id.btnFinalizar)
        rvEventos = findViewById(R.id.rvEventos)
        val btnVolver: TextView = findViewById(R.id.btnVolver)

        btnVolver.setOnClickListener { finish() }

        // Configurar RecyclerView
        eventoAdapter = EventoAdapter(dbManager.obtenerEventos())
        rvEventos.layoutManager = LinearLayoutManager(this)
        rvEventos.adapter = eventoAdapter

        // Configurar Autocompletado (HU-03)
        actualizarAdaptadorAutocompletado()

        // Restaurar evento en curso si se cerrÃ³ la app (HU-05)
        val prefs = getSharedPreferences(prefsName, MODE_PRIVATE)
        eventoEnCursoId = prefs.getLong(keyEventoId, -1L)
        if (eventoEnCursoId != -1L) {
            activarModoEnCurso(true)
        }

        // BotÃ³n Iniciar (HU-04, HU-05)
        btnIniciar.setOnClickListener {
            val nombreTarea = etBuscarTarea.text.toString().trim()
            if (nombreTarea.length < 2) {
                Toast.makeText(this, "Escribe al menos 2 letras para la tarea", Toast.LENGTH_SHORT).show()
                return@setOnClickListener
            }

            // Buscar si la tarea ya existe
            var tarea = dbManager.buscarTareaPorNombre(nombreTarea)
            if (tarea == null) {
                // Si no existe, pedir categorÃ­a (HU-04)
                mostrarDialogoCrearTarea(nombreTarea)
            } else {
                iniciarCronometroConTarea(tarea.id)
            }
        }

        // BotÃ³n Finalizar (HU-05)
        btnFinalizar.setOnClickListener {
            finalizarCronometroActual()
        }
    }

    private fun actualizarAdaptadorAutocompletado() {
        val tareas = dbManager.obtenerTareas()
        val nombresTareas = tareas.map { it.nombre }
        val adapter = ArrayAdapter(this, android.R.layout.simple_dropdown_item_1line, nombresTareas)
        etBuscarTarea.setAdapter(adapter)
    }

    private fun mostrarDialogoCrearTarea(nombreTarea: String) {
        val categorias = dbManager.obtenerCategorias()
        if (categorias.isEmpty()) {
            Toast.makeText(this, "No hay categorÃ­as registradas", Toast.LENGTH_SHORT).show()
            return
        }

        val nombresCategorias = categorias.map { it.nombre }.toTypedArray()
        var categoriaSeleccionadaIndex = 0

        AlertDialog.Builder(this)
            .setTitle("[ NUEVA TAREA ]")
            .setSingleChoiceItems(nombresCategorias, 0) { _, which ->
                categoriaSeleccionadaIndex = which
            }
            .setPositiveButton("Crear e Iniciar") { _, _ ->
                val catElegida = categorias[categoriaSeleccionadaIndex]
                val nuevaTareaId = dbManager.agregarTareaRetornandoId(nombreTarea, catElegida.id)
                if (nuevaTareaId != -1) {
                    actualizarAdaptadorAutocompletado()
                    iniciarCronometroConTarea(nuevaTareaId)
                } else {
                    Toast.makeText(this, "Error al crear la tarea", Toast.LENGTH_SHORT).show()
                }
            }
            .setNegativeButton("Cancelar", null)
            .show()
    }

    private fun iniciarCronometroConTarea(tareaId: Int) {
        val inicioIso = LocalDateTime.now().format(DateTimeFormatter.ISO_LOCAL_DATE_TIME)
        val idGenerado = dbManager.iniciarEvento(tareaId, inicioIso)

        if (idGenerado != -1L) {
            eventoEnCursoId = idGenerado
            getSharedPreferences(prefsName, MODE_PRIVATE).edit().putLong(keyEventoId, eventoEnCursoId).apply()
            activarModoEnCurso(true)
            Toast.makeText(this, "Evento iniciado correctamente", Toast.LENGTH_SHORT).show()
            eventoAdapter.updateData(dbManager.obtenerEventos())
        } else {
            Toast.makeText(this, "Error al iniciar el evento", Toast.LENGTH_SHORT).show()
        }
    }

    private fun finalizarCronometroActual() {
        val finIso = LocalDateTime.now().format(DateTimeFormatter.ISO_LOCAL_DATE_TIME)

        // Calcular duraciÃ³n obteniendo el evento de la base de datos
        val eventos = dbManager.obtenerEventos()
        val eventoActual = eventos.find { it.id.toLong() == eventoEnCursoId }

        var duracionTexto = "0m"
        if (eventoActual != null) {
            try {
                val inicioDt = LocalDateTime.parse(eventoActual.inicio, DateTimeFormatter.ISO_LOCAL_DATE_TIME)
                val finDt = LocalDateTime.parse(finIso, DateTimeFormatter.ISO_LOCAL_DATE_TIME)
                val duracion = Duration.between(inicioDt, finDt)
                val horas = duracion.toHours()
                val minutos = duracion.toMinutes() % 60
                duracionTexto = if (horas > 0) "${horas}h ${minutos}m" else "${minutos}m"
            } catch (e: Exception) {
                duracionTexto = "Completado"
            }
        }

        if (dbManager.finalizarEvento(eventoEnCursoId, finIso, duracionTexto)) {
            eventoEnCursoId = -1L
            getSharedPreferences(prefsName, MODE_PRIVATE).edit().remove(keyEventoId).apply()
            activarModoEnCurso(false)
            etBuscarTarea.setText("")
            Toast.makeText(this, "Evento finalizado con Ã©xito", Toast.LENGTH_SHORT).show()
            eventoAdapter.updateData(dbManager.obtenerEventos())
        } else {
            Toast.makeText(this, "Error al finalizar el evento", Toast.LENGTH_SHORT).show()
        }
    }

    private fun activarModoEnCurso(enCurso: Boolean) {
        btnIniciar.isEnabled = !enCurso
        btnIniciar.setBackgroundColor(if (enCurso) 0xFF444444.toInt() else 0xFFFFFFFF.toInt())

        btnFinalizar.isEnabled = enCurso
        btnFinalizar.setBackgroundColor(if (enCurso) 0xFFFFFFFF.toInt() else 0xFF444444.toInt())

        etBuscarTarea.isEnabled = !enCurso
    }
}

import android.widget.Button
import android.widget.TextView

class CategoriaAdapter(
    context: Context,
    private val categorias: List<Categoria>,
    private val onEditClick: (Categoria) -> Unit,
    private val onDeleteClick: (Categoria) -> Unit
) : ArrayAdapter<Categoria>(context, 0, categorias) {

    override fun getView(position: Int, convertView: View?, parent: ViewGroup): View {
        var view = convertView
        if (view == null) {
            view = LayoutInflater.from(context).inflate(R.layout.item_categoria, parent, false)
        }

        val categoria = categorias[position]

        val tvNombre = view!!.findViewById<TextView>(R.id.tvNombreCategoriaItem)
        val btnEditar = view.findViewById<Button>(R.id.btnEditarCategoria)
        val btnBorrar = view.findViewById<Button>(R.id.btnBorrarCategoria)

        tvNombre.text = categoria.nombre

        // Configuramos la acciÃ³n de los botones para que envÃ­en la categorÃ­a seleccionada a la Actividad
        btnEditar.setOnClickListener {
            onEditClick(categoria)
        }

        btnBorrar.setOnClickListener {
            onDeleteClick(categoria)
        }

        return view
    }
}

            cargarCategorias()

        }

            } else {
                Toast.makeText(this, "Error: Ya existe otra categorÃ­a con ese nombre", Toast.LENGTH_SHORT).show()

package pe.edu.instituto.ponce.mitiemposemanal

import android.content.ContentValues
import android.content.Context
import android.database.sqlite.SQLiteDatabase
import android.database.sqlite.SQLiteOpenHelper

class DatabaseHelper(context: Context) : SQLiteOpenHelper(context, DATABASE_NAME, null, DATABASE_VERSION) {

    companion object {
        private const val DATABASE_NAME = "MiTiempoSemanal.db"
        private const val DATABASE_VERSION = 2

        const val TABLE_CATEGORIES = "categorias"
        const val COL_CAT_ID = "id"
        const val COL_CAT_NAME = "nombre"

        const val TABLE_TASKS = "tareas"
        const val COL_TASK_ID = "id"
        const val COL_TASK_NAME = "nombre"
        const val COL_TASK_CAT_ID = "categoria_id"

        // Tabla Eventos actualizada para HU-05 y Sprint 3
        const val TABLE_EVENTS = "eventos"
        const val COL_EV_ID = "id"
        const val COL_EV_TASK_ID = "tarea_id"
        const val COL_EV_START = "inicio" // LocalDateTime en ISO
        const val COL_EV_END = "fin" // LocalDateTime en ISO
        const val COL_EV_DURATION = "duracion" // DuraciÃ³n en formato texto o minutos
    }

    override fun onCreate(db: SQLiteDatabase) {
        val createCategoriesTable = ("CREATE TABLE $TABLE_CATEGORIES ("
                + "$COL_CAT_ID INTEGER PRIMARY KEY AUTOINCREMENT,"
                + "$COL_CAT_NAME TEXT UNIQUE NOT NULL)")
        db.execSQL(createCategoriesTable)

        val createTasksTable = ("CREATE TABLE $TABLE_TASKS ("
                + "$COL_TASK_ID INTEGER PRIMARY KEY AUTOINCREMENT,"
                + "$COL_TASK_NAME TEXT NOT NULL,"
                + "$COL_TASK_CAT_ID INTEGER NOT NULL,"
                + "FOREIGN KEY($COL_TASK_CAT_ID) REFERENCES $TABLE_CATEGORIES($COL_CAT_ID) ON DELETE CASCADE)")
        db.execSQL(createTasksTable)

        val createEventsTable = ("CREATE TABLE $TABLE_EVENTS ("
                + "$COL_EV_ID INTEGER PRIMARY KEY AUTOINCREMENT,"
                + "$COL_EV_TASK_ID INTEGER NOT NULL,"
                + "$COL_EV_START TEXT NOT NULL,"
                + "$COL_EV_END TEXT,"
                + "$COL_EV_DURATION TEXT,"
                + "FOREIGN KEY($COL_EV_TASK_ID) REFERENCES $TABLE_TASKS($COL_TASK_ID) ON DELETE CASCADE)")
        db.execSQL(createEventsTable)

        insertarCategoriaDefecto(db, "Estudio")
        insertarCategoriaDefecto(db, "Trabajo")
        insertarCategoriaDefecto(db, "Descanso")
        insertarCategoriaDefecto(db, "Ocio")
    }

    private fun insertarCategoriaDefecto(db: SQLiteDatabase, nombre: String) {
        val values = ContentValues()
        values.put(COL_CAT_NAME, nombre)
        db.insert(TABLE_CATEGORIES, null, values)
    }

    override fun onUpgrade(db: SQLiteDatabase, oldVersion: Int, newVersion: Int) {
        if (oldVersion < 2) {
            val createEventsTable = ("CREATE TABLE $TABLE_EVENTS ("
                    + "$COL_EV_ID INTEGER PRIMARY KEY AUTOINCREMENT,"
                    + "$COL_EV_TASK_ID INTEGER NOT NULL,"
                    + "$COL_EV_START TEXT NOT NULL,"
                    + "$COL_EV_END TEXT,"
                    + "$COL_EV_DURATION TEXT,"
                    + "FOREIGN KEY($COL_EV_TASK_ID) REFERENCES $TABLE_TASKS($COL_TASK_ID) ON DELETE CASCADE)")
            db.execSQL(createEventsTable)
        }
    }

    override fun onOpen(db: SQLiteDatabase) {
        super.onOpen(db)
        if (!db.isReadOnly) {
            db.execSQL("PRAGMA foreign_keys=ON;")
        }
    }
}

package pe.edu.instituto.ponce.mitiemposemanal

import android.content.ContentValues
import android.content.Context

class DatabaseManager(context: Context) {
    private val dbHelper = DatabaseHelper(context)

    // ================= MÃ‰TODOS PARA CATEGORÃAS =================

    fun obtenerCategorias(): List<Categoria> {
        val lista = mutableListOf<Categoria>()
        val db = dbHelper.readableDatabase
        val cursor = db.rawQuery("SELECT * FROM ${DatabaseHelper.TABLE_CATEGORIES} ORDER BY ${DatabaseHelper.COL_CAT_NAME}", null)
        if (cursor.moveToFirst()) {
            do {
                val id = cursor.getInt(cursor.getColumnIndexOrThrow(DatabaseHelper.COL_CAT_ID))
                val nombre = cursor.getString(cursor.getColumnIndexOrThrow(DatabaseHelper.COL_CAT_NAME))
                lista.add(Categoria(id, nombre))
            } while (cursor.moveToNext())
        }
        cursor.close()
        db.close()
        return lista
    }

    fun existeCategoria(nombre: String, idAignorar: Int = -1): Boolean {
        val db = dbHelper.readableDatabase
        val query = "SELECT COUNT(*) FROM ${DatabaseHelper.TABLE_CATEGORIES} WHERE LOWER(${DatabaseHelper.COL_CAT_NAME}) = LOWER(?) AND ${DatabaseHelper.COL_CAT_ID} != ?"
        val cursor = db.rawQuery(query, arrayOf(nombre, idAignorar.toString()))
        var existe = false
        if (cursor.moveToFirst()) {
            existe = cursor.getInt(0) > 0
        }
        cursor.close()
        db.close()
        return existe
    }

    fun agregarCategoria(nombre: String): Boolean {
        if (existeCategoria(nombre)) return false
        val db = dbHelper.writableDatabase
        val values = ContentValues().apply {
            put(DatabaseHelper.COL_CAT_NAME, nombre)
        }
        val result = db.insert(DatabaseHelper.TABLE_CATEGORIES, null, values)
        db.close()
        return result != -1L
    }

    fun actualizarCategoria(id: Int, nombre: String): Boolean {
        if (existeCategoria(nombre, id)) return false
        val db = dbHelper.writableDatabase
        val values = ContentValues().apply {
            put(DatabaseHelper.COL_CAT_NAME, nombre)
        }
        val result = db.update(DatabaseHelper.TABLE_CATEGORIES, values, "${DatabaseHelper.COL_CAT_ID}=?", arrayOf(id.toString()))
        db.close()
        return result > 0
    }

    fun categoriaTieneTareas(idCategoria: Int): Boolean {
        val db = dbHelper.readableDatabase
        val query = "SELECT COUNT(*) FROM ${DatabaseHelper.TABLE_TASKS} WHERE ${DatabaseHelper.COL_TASK_CAT_ID} = ?"
        val cursor = db.rawQuery(query, arrayOf(idCategoria.toString()))
        var tieneTareas = false
        if (cursor.moveToFirst()) {
            tieneTareas = cursor.getInt(0) > 0
        }
        cursor.close()
        db.close()
        return tieneTareas
    }

    fun eliminarCategoria(id: Int): Boolean {
        if (categoriaTieneTareas(id)) return false
        val db = dbHelper.writableDatabase
        val result = db.delete(DatabaseHelper.TABLE_CATEGORIES, "${DatabaseHelper.COL_CAT_ID}=?", arrayOf(id.toString()))
        db.close()
        return result > 0
    }

    // ================= MÃ‰TODOS PARA TAREAS =================

    fun obtenerTareas(): List<Tarea> {
        val lista = mutableListOf<Tarea>()
        val db = dbHelper.readableDatabase
        val query = """
            SELECT t.${DatabaseHelper.COL_TASK_ID}, t.${DatabaseHelper.COL_TASK_NAME}, 
                   t.${DatabaseHelper.COL_TASK_CAT_ID}, c.${DatabaseHelper.COL_CAT_NAME}
            FROM ${DatabaseHelper.TABLE_TASKS} t
            INNER JOIN ${DatabaseHelper.TABLE_CATEGORIES} c ON t.${DatabaseHelper.COL_TASK_CAT_ID} = c.${DatabaseHelper.COL_CAT_ID}
            ORDER BY t.${DatabaseHelper.COL_TASK_NAME}
        """.trimIndent()

        val cursor = db.rawQuery(query, null)
        if (cursor.moveToFirst()) {
            do {
                lista.add(Tarea(cursor.getInt(0), cursor.getString(1), cursor.getInt(2), cursor.getString(3)))
            } while (cursor.moveToNext())
        }
        cursor.close()
        db.close()
        return lista
    }

    fun existeTarea(nombre: String, idAignorar: Int = -1): Boolean {
        val db = dbHelper.readableDatabase
        val query = "SELECT COUNT(*) FROM ${DatabaseHelper.TABLE_TASKS} WHERE LOWER(${DatabaseHelper.COL_TASK_NAME}) = LOWER(?) AND ${DatabaseHelper.COL_TASK_ID} != ?"
        val cursor = db.rawQuery(query, arrayOf(nombre, idAignorar.toString()))
        var existe = false
        if (cursor.moveToFirst()) {
            existe = cursor.getInt(0) > 0
        }
        cursor.close()
        db.close()
        return existe
    }

    fun agregarTarea(nombre: String, idCategoria: Int): Boolean {
        if (existeTarea(nombre)) return false
        val db = dbHelper.writableDatabase
        val values = ContentValues().apply {
            put(DatabaseHelper.COL_TASK_NAME, nombre.trim())
            put(DatabaseHelper.COL_TASK_CAT_ID, idCategoria)
        }
        val result = db.insert(DatabaseHelper.TABLE_TASKS, null, values)
        db.close()
        return result != -1L
    }

    fun agregarTareaRetornandoId(nombre: String, idCategoria: Int): Int {
        val db = dbHelper.writableDatabase
        val values = ContentValues().apply {
            put(DatabaseHelper.COL_TASK_NAME, nombre.trim())
            put(DatabaseHelper.COL_TASK_CAT_ID, idCategoria)
        }
        val id = db.insert(DatabaseHelper.TABLE_TASKS, null, values).toInt()
        db.close()
        return id
    }

    fun actualizarTarea(id: Int, nombre: String, idCategoria: Int): Boolean {
        if (existeTarea(nombre, id)) return false
        val db = dbHelper.writableDatabase
        val values = ContentValues().apply {
            put(DatabaseHelper.COL_TASK_NAME, nombre.trim())
            put(DatabaseHelper.COL_TASK_CAT_ID, idCategoria)
        }
        val result = db.update(DatabaseHelper.TABLE_TASKS, values, "${DatabaseHelper.COL_TASK_ID}=?", arrayOf(id.toString()))
        db.close()
        return result > 0
    }

    fun tareaTieneEventos(idTarea: Int): Boolean {
        val db = dbHelper.readableDatabase
        val query = "SELECT COUNT(*) FROM ${DatabaseHelper.TABLE_EVENTS} WHERE ${DatabaseHelper.COL_EV_TASK_ID} = ?"
        val cursor = db.rawQuery(query, arrayOf(idTarea.toString()))
        var tieneEventos = false
        if (cursor.moveToFirst()) {
            tieneEventos = cursor.getInt(0) > 0
        }
        cursor.close()
        db.close()
        return tieneEventos
    }

    fun eliminarTarea(id: Int): Boolean {
        if (tareaTieneEventos(id)) return false
        val db = dbHelper.writableDatabase
        val result = db.delete(DatabaseHelper.TABLE_TASKS, "${DatabaseHelper.COL_TASK_ID}=?", arrayOf(id.toString()))
        db.close()
        return result > 0
    }

    fun buscarTareaPorNombre(nombre: String): Tarea? {
        val db = dbHelper.readableDatabase
        val query = """
            SELECT t.${DatabaseHelper.COL_TASK_ID}, t.${DatabaseHelper.COL_TASK_NAME}, 
                   t.${DatabaseHelper.COL_TASK_CAT_ID}, c.${DatabaseHelper.COL_CAT_NAME}
            FROM ${DatabaseHelper.TABLE_TASKS} t
            INNER JOIN ${DatabaseHelper.TABLE_CATEGORIES} c ON t.${DatabaseHelper.COL_TASK_CAT_ID} = c.${DatabaseHelper.COL_CAT_ID}
            WHERE LOWER(t.${DatabaseHelper.COL_TASK_NAME}) = LOWER(?)
        """.trimIndent()
        val cursor = db.rawQuery(query, arrayOf(nombre.trim()))
        var tarea: Tarea? = null
        if (cursor.moveToFirst()) {
            tarea = Tarea(cursor.getInt(0), cursor.getString(1), cursor.getInt(2), cursor.getString(3))
        }
        cursor.close()
        db.close()
        return tarea
    }

    // ================= MÃ‰TODOS PARA EVENTOS (SPRINT 2) =================

    fun iniciarEvento(tareaId: Int, inicioIso: String): Long {
        val db = dbHelper.writableDatabase
        val values = ContentValues().apply {
            put(DatabaseHelper.COL_EV_TASK_ID, tareaId)
            put(DatabaseHelper.COL_EV_START, inicioIso)
        }
        val id = db.insert(DatabaseHelper.TABLE_EVENTS, null, values)
        db.close()
        return id
    }

    fun finalizarEvento(eventoId: Long, finIso: String, duracionStr: String): Boolean {
        val db = dbHelper.writableDatabase
        val values = ContentValues().apply {
            put(DatabaseHelper.COL_EV_END, finIso)
            put(DatabaseHelper.COL_EV_DURATION, duracionStr)
        }
        val result = db.update(DatabaseHelper.TABLE_EVENTS, values, "${DatabaseHelper.COL_EV_ID}=?", arrayOf(eventoId.toString()))
        db.close()
        return result > 0
    }

    fun obtenerEventos(): List<Evento> {
        val lista = mutableListOf<Evento>()
        val db = dbHelper.readableDatabase
        val query = """
            SELECT e.${DatabaseHelper.COL_EV_ID}, e.${DatabaseHelper.COL_EV_TASK_ID}, 
                   t.${DatabaseHelper.COL_TASK_NAME}, c.${DatabaseHelper.COL_CAT_NAME},
                   e.${DatabaseHelper.COL_EV_START}, e.${DatabaseHelper.COL_EV_END}, e.${DatabaseHelper.COL_EV_DURATION}
            FROM ${DatabaseHelper.TABLE_EVENTS} e
            INNER JOIN ${DatabaseHelper.TABLE_TASKS} t ON e.${DatabaseHelper.COL_EV_TASK_ID} = t.${DatabaseHelper.COL_TASK_ID}
            INNER JOIN ${DatabaseHelper.TABLE_CATEGORIES} c ON t.${DatabaseHelper.COL_TASK_CAT_ID} = c.${DatabaseHelper.COL_CAT_ID}
            ORDER BY e.${DatabaseHelper.COL_EV_ID} DESC
        """.trimIndent()

        val cursor = db.rawQuery(query, null)
        if (cursor.moveToFirst()) {
            do {
                lista.add(
                    Evento(
                        id = cursor.getInt(0),
                        tareaId = cursor.getInt(1),
                        tareaNombre = cursor.getString(2),
                        categoriaNombre = cursor.getString(3),
                        inicio = cursor.getString(4),
                        fin = cursor.getString(5),
                        duracion = cursor.getString(6)
                    )
                )
            } while (cursor.moveToNext())
        }
        cursor.close()
        db.close()
        return lista
    }

    private val outputTimeFormatter = DateTimeFormatter.ofPattern("dd/MM HH:mm")

    class EventoViewHolder(view: View) : RecyclerView.ViewHolder(view) {
        val tvTarea: TextView = view.findViewById(R.id.tvItemTarea)
        val tvCategoria: TextView = view.findViewById(R.id.tvItemCategoria)

        return EventoViewHolder(view)
    }

        val evento = eventos[position]
        holder.tvTarea.text = evento.tareaNombre
        holder.tvCategoria.text = "CategorÃ­a: ${evento.categoriaNombre}"

        try {
            val inicioParsed = LocalDateTime.parse(evento.inicio, inputFormatter)
            val inicioStr = inicioParsed.format(outputTimeFormatter)

            if (evento.fin != null) {
                val finParsed = LocalDateTime.parse(evento.fin, inputFormatter)
                val finStr = finParsed.format(outputTimeFormatter)
                holder.tvHorarios.text = "Inicio: $inicioStr  â€¢  Fin: $finStr"
                holder.tvDuracion.text = evento.duracion ?: ""
            } else {
                holder.tvHorarios.text = "Inicio: $inicioStr  â€¢  [EN CURSO]"
                holder.tvDuracion.text = "Activo"
            }
        } catch (e: Exception) {
            holder.tvHorarios.text = "Inicio: ${evento.inicio}"
            holder.tvDuracion.text = evento.duracion ?: ""
        }
    }

    override fun getItemCount(): Int = eventos.size

    fun updateData(nuevosEventos: List<Evento>) {
        eventos = nuevosEventos
        notifyDataSetChanged()
    }

class TareasActivity : AppCompatActivity() {

    private lateinit var dbManager: DatabaseManager
    private lateinit var etNombreTarea: EditText
    private lateinit var spCategorias: Spinner
    private lateinit var btnGuardarTarea: Button
    private lateinit var lvTareas: ListView
    private lateinit var btnVolverTareas: Button

    private var listaCategorias = listOf<Categoria>()
    private var listaTareas = listOf<Tarea>()

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_tareas)

        dbManager = DatabaseManager(this)

        etNombreTarea = findViewById(R.id.etNombreTarea)
        spCategorias = findViewById(R.id.spCategorias)
        btnGuardarTarea = findViewById(R.id.btnGuardarTarea)
        lvTareas = findViewById(R.id.lvTareas)
        btnVolverTareas = findViewById(R.id.btnVolverTareas)

        cargarCategorias()
        cargarTareas()

        // El botÃ³n superior ahora solo sirve para CREAR
        btnGuardarTarea.setOnClickListener {
            guardarNuevaTarea()
        }

        btnVolverTareas.setOnClickListener {
            finish()
        }
    }

    private fun cargarCategorias() {
        listaCategorias = dbManager.obtenerCategorias()
        val adapter = ArrayAdapter(this, android.R.layout.simple_spinner_dropdown_item, listaCategorias)
        spCategorias.adapter = adapter
    }

    private fun cargarTareas() {
        listaTareas = dbManager.obtenerTareas()
        val adapter = TareaAdapter(
            this,
            listaTareas,
            onEditClick = { tarea -> mostrarDialogoEditar(tarea) },
            onDeleteClick = { tarea -> mostrarDialogoEliminar(tarea) }
        )
        lvTareas.adapter = adapter
    }

    private fun guardarNuevaTarea() {
        val nombre = etNombreTarea.text.toString().trim()
        val categoriaSeleccionada = spCategorias.selectedItem as? Categoria

        if (nombre.isEmpty()) {
            Toast.makeText(this, "La tarea debe tener un nombre", Toast.LENGTH_SHORT).show()
            return
        }

        if (categoriaSeleccionada == null) {
            Toast.makeText(this, "Debes crear y seleccionar una categorÃ­a primero", Toast.LENGTH_SHORT).show()
            return
        }

        val exito = dbManager.agregarTarea(nombre, categoriaSeleccionada.id)
        if (exito) {
            Toast.makeText(this, "Tarea guardada", Toast.LENGTH_SHORT).show()
            etNombreTarea.text.clear()
            if (listaCategorias.isNotEmpty()) spCategorias.setSelection(0)
            cargarTareas()
        } else {
            Toast.makeText(this, "Error: Ya existe una tarea con ese nombre", Toast.LENGTH_SHORT).show()
        }
    }

    private fun mostrarDialogoEditar(tarea: Tarea) {
        val dialog = Dialog(this)
        dialog.setContentView(R.layout.dialog_editar_tarea)

        val etEditarNombre = dialog.findViewById<EditText>(R.id.etEditarNombreTarea)
        val spEditarCategoria = dialog.findViewById<Spinner>(R.id.spEditarCategoriaTarea)
        val btnCancelar = dialog.findViewById<Button>(R.id.btnCancelarEditarTarea)
        val btnConfirmar = dialog.findViewById<Button>(R.id.btnConfirmarEditarTarea)

        etEditarNombre.setText(tarea.nombre)

        val adapter = ArrayAdapter(this, android.R.layout.simple_spinner_dropdown_item, listaCategorias)
        spEditarCategoria.adapter = adapter

        val indexCategoria = listaCategorias.indexOfFirst { it.id == tarea.categoriaId }
        if (indexCategoria != -1) {
            spEditarCategoria.setSelection(indexCategoria)
        }

        btnCancelar.setOnClickListener { dialog.dismiss() }
        btnConfirmar.setOnClickListener {
            val nuevoNombre = etEditarNombre.text.toString().trim()
            val categoriaSeleccionada = spEditarCategoria.selectedItem as? Categoria

            if (nuevoNombre.isEmpty()) {
                Toast.makeText(this, "La tarea debe tener un nombre", Toast.LENGTH_SHORT).show()
                return@setOnClickListener
            }

            if (categoriaSeleccionada == null) return@setOnClickListener

            val exito = dbManager.actualizarTarea(tarea.id, nuevoNombre, categoriaSeleccionada.id)
            if (exito) {
                Toast.makeText(this, "Tarea actualizada", Toast.LENGTH_SHORT).show()
                cargarTareas()
                dialog.dismiss()
            } else {
                Toast.makeText(this, "Error: Ya existe otra tarea con ese nombre", Toast.LENGTH_SHORT).show()
            }
        }
        dialog.show()
    }

    private fun mostrarDialogoEliminar(tarea: Tarea) {
        val dialog = Dialog(this)
        dialog.setContentView(R.layout.dialog_eliminar)

        val tvMensaje = dialog.findViewById<TextView>(R.id.tvMensajeEliminar)
        val btnCancelar = dialog.findViewById<Button>(R.id.btnCancelarEliminar)
        val btnConfirmar = dialog.findViewById<Button>(R.id.btnConfirmarEliminar)

        tvMensaje.text = "Â¿Eliminar la tarea '${tarea.nombre}'?"

        btnCancelar.setOnClickListener { dialog.dismiss() }
        btnConfirmar.setOnClickListener {
            val exito = dbManager.eliminarTarea(tarea.id)
            if (exito) {
                Toast.makeText(this, "Tarea eliminada", Toast.LENGTH_SHORT).show()
                cargarTareas()
            } else {
                Toast.makeText(this, "Error al eliminar", Toast.LENGTH_SHORT).show()
            }
            dialog.dismiss()
        }
        dialog.show()
    }
}
